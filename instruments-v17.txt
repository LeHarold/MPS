function visibleSnapPoints(){return [...waypointPicker.getVisiblePoints().map(p=>({name:p.name,position:p.position})),...state.layers.filter(l=>l.visible&&(l.opacity??1)>0).flatMap(l=>l.objects.filter(o=>o.type==='point').map(o=>({name:o.name,position:o.points[0]})))]}
function originPixel(name){const g=instrumentGeometry(name);return name==='ruler'?instrumentPoint(g,18,20):[g.cx,g.cy]}
function endPixel(name){const g=instrumentGeometry(name);return name==='ruler'?instrumentPoint(g,g.width-18,20):instrumentPoint(g,g.width/2,0)}
function applyInstrumentGeometry(name,g){const s=state.instruments[name],size=map.getSize();s.x=g.cx/size.x;s.y=g.cy/size.y;s.rotation=g.rotation;s.scale=g.width/instrumentSizes[name].width}
function lockOrigin(name,point){state.instruments[name].lock={point:[point[0],point[1]]};renderInstruments()}
function snapOrigin(name){if(state.settings.rulerSnap===false)return false;const origin=originPixel(name);const nearest=visibleSnapPoints().map(p=>{const q=map.latLngToContainerPoint(p.position);return{...p,pixel:[q.x,q.y],d:Math.hypot(q.x-origin[0],q.y-origin[1])}}).filter(p=>p.d<24).sort((a,b)=>a.d-b.d)[0];if(!nearest)return false;lockOrigin(name,nearest.position);toast('0 點已吸附並鎖定 '+nearest.name+'；點擊 0 解鎖');return true}
function snapRulerBody(){if(state.settings.rulerSnap===false)return;const s=state.instruments.ruler,a=originPixel('ruler'),b=endPixel('ruler');const candidate=visibleSnapPoints().map(p=>{const pixel=map.latLngToContainerPoint(p.position),q=[pixel.x,pixel.y],contact=rulerContact(a,b,q);return{...p,q,contact}}).filter(p=>p.contact&&p.contact.distance>25).sort((a,b)=>a.contact.error-b.contact.error)[0];if(!candidate){s.contact=null;return}const c=candidate.contact,rotation=Math.atan2(candidate.q[1]-a[1],candidate.q[0]-a[0])*180/Math.PI;const g=instrumentGeometry('ruler');applyInstrumentGeometry('ruler',{width:g.width,rotation,...positionFromAnchor('ruler',a,g.width,90,rotation)});s.contact={fraction:Math.min(1,c.distance/(g.width-36)),name:candidate.name};renderInstruments()}
function settleInstrument(name){const s=state.instruments[name];if(!s.lock)snapOrigin(name);if(name==='ruler')snapRulerBody();else if(!s.lock){const before=s.rotation;snapProtractor();if(s.rotation!==before){const at=map.containerPointToLatLng(originPixel(name));lockOrigin(name,[at.lat,at.lng])}}save()}

 el.addEventListener('pointerdown',e=>{if(tool==='free'&&!e.target.closest('[data-instrument]')||e.pointerType==='mouse'&&e.button!==0)return;e.preventDefault();e.stopPropagation();const s=state.instruments[name];if(e.target.closest('[data-zero]')&&s.lock){snapshot();s.lock=null;s.contact=null;renderInstruments();save('0 點已解除鎖定');return}if(!pointers.size)snapshot();pointers.set(e.pointerId,{x:e.clientX,y:e.clientY});el.setPointerCapture(e.pointerId);gestures.set(name,true);begin()});
 el.addEventListener('pointermove',e=>{if(!pointers.has(e.pointerId)||!baseline)return;e.preventDefault();pointers.set(e.pointerId,{x:e.clientX,y:e.clientY});const ps=[...pointers.values()],size=map.getSize(),s=state.instruments[name];if(s.lock){const anchor=map.latLngToContainerPoint(s.lock.point);if(ps.length>=2&&baseline.mid){s.rotation=baseline.rotation+angleDelta(gestureAngle(ps[0],ps[1]),baseline.angle);s.scale=Math.max(.2,Math.min(3,baseline.scale*Math.hypot(ps[1].x-ps[0].x,ps[1].y-ps[0].y)/Math.max(1,baseline.distance)))}else{const desired=[baseline.end[0]+e.clientX-baseline.p.x,baseline.end[1]+e.clientY-baseline.p.y],g=resizeFromAnchor(name,[anchor.x,anchor.y],desired,Math.min(size.x*2,instrumentSizes[name].width*3));applyInstrumentGeometry(name,g)}renderInstruments();if(name==='ruler')snapRulerBody()}else if(ps.length>=2&&baseline.mid){const mid={x:(ps[0].x+ps[1].x)/2,y:(ps[0].y+ps[1].y)/2};s.x=Math.max(0,Math.min(1,baseline.x+(mid.x-baseline.mid.x)/size.x));s.y=Math.max(0,Math.min(.88,baseline.y+(mid.y-baseline.mid.y)/size.y));s.rotation=baseline.rotation+angleDelta(gestureAngle(ps[0],ps[1]),baseline.angle);s.scale=Math.max(.2,Math.min(3,baseline.scale*Math.hypot(ps[1].x-ps[0].x,ps[1].y-ps[0].y)/Math.max(1,baseline.distance)))}else{s.x=Math.max(0,Math.min(1,baseline.x+(e.clientX-baseline.p.x)/size.x));s.y=Math.max(0,Math.min(.88,baseline.y+(e.clientY-baseline.p.y)/size.y))}renderInstruments();updateInstrumentControls()});
