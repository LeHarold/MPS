function renderAxes(){const size=map.getSize(),height=size.y-93;if(size.x<100||height<30)return;const bounds=map.getBounds();for(const lat of [false,true]){const axis=$(lat?'latitude-axis':'longitude-axis');axis.replaceChildren();const lo=lat?bounds.getSouth():bounds.getWest(),hi=lat?bounds.getNorth():bounds.getEast(),span=hi-lo,pixels=lat?height:size.x,pxPerDegree=pixels/span;const step=pxPerDegree>=90?1/60:pxPerDegree>=25?5/60:pxPerDegree>=10?15/60:1;const labels=[];let ticks=0;for(let v=Math.ceil(lo/step)*step;v<=hi&&ticks++<4000;v+=step){const pixel=map.latLngToContainerPoint(lat?[v,map.getCenter().lng]:[map.getCenter().lat,v]),position=lat?pixel.y:pixel.x;if(position<10||position>(lat?height-5:size.x-12))continue;const minute=Math.round(v*60),degree=Math.abs(minute%60)===0,tick=document.createElement('i');tick.className='axis-tick '+(lat?'lat-tick':'lng-tick')+(degree?' degree-tick':minute%5===0?' five-minute-tick':'');tick.style[lat?'top':'left']=position+'px';axis.append(tick);if(degree||pxPerDegree>=120&&(minute%5===0||pxPerDegree>=900)){const text=degree?axisFormat(v,lat,1):axisFormat(v,lat,1/60);labels.push({position,text,extent:lat?15:text.length*6+4,priority:degree?2:1})}}if(!labels.length){const step=axisStep(span,lat?Math.floor(height/45):Math.floor(size.x/120));for(let v=Math.ceil(lo/step)*step;v<=hi;v+=step){const p=map.latLngToContainerPoint(lat?[v,map.getCenter().lng]:[map.getCenter().lat,v]),text=axisFormat(v,lat,step);labels.push({position:lat?p.y:p.x,text,extent:lat?15:text.length*6+4,priority:0})}}for(const item of labelLayout(labels,lat?10:8)){const el=document.createElement('span');el.className='degree-label '+(lat?'degree-lat':'degree-lng');el.textContent=item.text;el.style[lat?'top':'left']=item.position+'px';axis.append(el)}}}
