function constrainedName(label){const el=label.getElement?.(),classes=label.options?.className||'',text=el?.textContent||'';return !!el&&!label.areaPoints&&!/cursor-readout|object-coordinate-label|area-name-label/.test(classes)&&label.options?.mpsCategory!=='distance'&&!el.dataset.distanceDrag&&!/\b(?:NM|KM|M)\b/.test(text)}
function nameRadiusPixels(label,ownerMap){const at=label.getLatLng(),anchor=ownerMap.latLngToContainerPoint(at),values=[0,90,180,270].map(angle=>ownerMap.latLngToContainerPoint(destination([at.lat,at.lng],angle,3*1852)).distanceTo(anchor));return Math.max(0,Math.min(...values.filter(Number.isFinite))||0)}
function constrainNameOffset(label,ownerMap,offset,rect=null,rootRect=null){if(!constrainedName(label))return offset;const at=label.getLatLng();if(!at)return offset;if(!rect){label.options.offset=offset;label.update();rect=label.getElement().getBoundingClientRect()}if(!rect.width||!rect.height)return offset;const root=rootRect||ownerMap.getContainer().getBoundingClientRect(),anchor=ownerMap.latLngToContainerPoint(at),x=Math.max(rect.left-root.left,Math.min(rect.right-root.left,anchor.x)),y=Math.max(rect.top-root.top,Math.min(rect.bottom-root.top,anchor.y)),dx=x-anchor.x,dy=y-anchor.y,gap=Math.hypot(dx,dy);if(!gap)return offset;let fraction=Math.min(1,nameRadiusPixels(label,ownerMap)/gap);const measure=t=>{const p=ownerMap.containerPointToLatLng([anchor.x+dx*t,anchor.y+dy*t]);return distance([at.lat,at.lng],[p.lat,p.lng])};if(measure(fraction)>3*1852){let low=0,high=fraction;for(let i=0;i<24;i++){const t=(low+high)/2;if(measure(t)<=3*1852)low=t;else high=t}fraction=low}return L.point(offset.x+dx*(fraction-1),offset.y+dy*(fraction-1))}
function prepareNameConstraints(ownerMap,root){const bounds=ownerMap.getContainer().getBoundingClientRect();ownerMap.eachLayer(label=>{if(!(label instanceof L.Tooltip)||!label.options.permanent||!constrainedName(label))return;const el=label.getElement(),at=label.getLatLng();if(!el||!at)return;const old=L.point(label.options.offset||[0,0]),offset=constrainNameOffset(label,ownerMap,old,el.getBoundingClientRect(),bounds);label.options.offset=offset;label.update();const base=label.options.mpsBaseOffset||[0,0],manual=state.settings.labelPositions?.[label.options.mpsLabelKey]||{x:0,y:0};label.options.mpsAutoOffset=[offset.x-base[0]-manual.x,offset.y-base[1]-manual.y];const anchor=ownerMap.latLngToContainerPoint(at);el.dataset.mpsAnchorX=anchor.x;el.dataset.mpsAnchorY=anchor.y;el.dataset.mpsMaxGap=nameRadiusPixels(label,ownerMap)})}
map.on('tooltipopen',event=>{const label=event.tooltip;if(label&&!label.options.permanent&&constrainedName(label)){label.options.offset=constrainNameOffset(label,map,L.point(label.options.offset||[0,0]));label.update()}});
