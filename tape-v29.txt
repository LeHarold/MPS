function renderTapeDraft(cursor){draft.clearLayers();if(!tapeStart)return;L.circleMarker(tapeStart.point,{radius:5,color:state.settings.color,fillOpacity:.8,interactive:false}).addTo(draft);if(cursor){const p=[cursor.lat,cursor.lng,0];L.polyline([tapeStart.point,p],{color:state.settings.color,weight:2,dashArray:'5 4',interactive:false}).bindTooltip(formatDistance(distance(tapeStart.point,p)),{permanent:true,direction:'center'}).addTo(draft)}$('drawing-status').textContent='鐵捲尺：請選第二點'}
function pickTape(point,name=''){const p=[point[0],point[1],point[2]||0];if(!tapeStart){tapeStart={point:p,name};renderTapeDraft();toast('鐵捲尺：已選起點，請點選第二點');return}if(distance(tapeStart.point,p)<.01)return toast('請選取與起點不同的第二點');const start=tapeStart,number=state.layers.flatMap(l=>l.objects).filter(o=>o.tapeMeasure).length+1,o=addObject('line',[start.point,p],'鐵捲尺 '+number,'起點：'+(start.name||pair(start.point))+'\n終點：'+(name||pair(p)));o.measureDistance=true;o.tapeMeasure=true;tapeStart=null;draft.clearLayers();draftPoints=[];renderObjects();renderLayers();save('已建立測距直線：'+formatDistance(distance(start.point,p)));$('drawing-status').textContent='鐵捲尺：請選第一點'}
