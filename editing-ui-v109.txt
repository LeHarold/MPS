function objectMoveTarget(id,host,vertexIndex=null){const found=getObject(id);if(!found)return null;const layer=found.layer,grouped=layer.moveAsGroup===true,objects=grouped?layer.objects:[found.object];const pivot=()=>{if(vertexIndex!==null&&!grouped)return found.object.points[vertexIndex];const pts=objects.flatMap(o=>o.points),origin=pts[0][1],lngs=pts.map(p=>((p[1]-origin+540)%360)-180+origin);return[(Math.min(...pts.map(p=>p[0]))+Math.max(...pts.map(p=>p[0])))/2,((Math.min(...lngs)+Math.max(...lngs))/2+540)%360-180,0]};
 return{name:grouped?'群組：'+layer.name:found.object.name,rotate:vertexIndex===null||grouped,scale:(vertexIndex===null||grouped)&&!objects.some(o=>o.distanceLocked),anchor:()=>localAnchor(map,pivot(),host),capture(){return{origin:[...pivot()],objects:objects.map(o=>clone(o))}},apply(delta,base,mode){const at=map.latLngToContainerPoint(base.origin),next=mode==='move'?map.containerPointToLatLng([at.x+delta.dx,at.y+delta.dy]):L.latLng(base.origin[0],base.origin[1]);for(const original of base.objects){const object=getObject(original.id)?.object;if(!object)continue;const angle=mode==='rotate'?delta.angle:0,scale=mode==='scale'?delta.scale:1;if(vertexIndex!==null&&!grouped){object.points[vertexIndex]=transformPoints([original.points[vertexIndex]],base.origin,[next.lat,next.lng],0,1)[0];if(object.distanceLocked&&object.points.length===2){const other=1-vertexIndex;object.points[vertexIndex]=destination(object.points[other],bearing(object.points[other],object.points[vertexIndex]),object.lockedDistanceM,object.points[vertexIndex][2])}}else{object.points=transformPoints(original.points,base.origin,[next.lat,next.lng],angle,scale);if(original.type==='point'&&mode==='scale')object.pointScale=Math.max(.05,Math.min(4,(original.pointScale??1)*scale));if(original.compassArc){object.compassArc={...original.compassArc,center:transformPoints([original.compassArc.center],base.origin,[next.lat,next.lng],angle,scale)[0],radiusM:original.compassArc.radiusM*scale}}}if(object.boxInputMode==='ordered'&&object.boxVertices)object.boxVertices.forEach((v,i)=>v.position=[...object.points[i]])}applyPointMOMs(state);renderObjects();renderLabelConnections(map)},finish(){renderLayers();if(selected&&getObject()){showHandles(getObject().object);loadDetailPoints(getObject().object)}}};
}
