
// A compass pinch changes its radius and heading, never draws an accidental arc.
const compassTouches=new Map();let compassTouchSession=null,compassPinch=null,compassIgnoreClickUntil=0;
const compassTouchHost=map.getContainer();
compassTouchHost.addEventListener('pointerdown',e=>{if(e.pointerType!=='touch'||!compassState()?.visible)return;const target=e.target.closest?.('.compass-center,#compass-end,.compass-gesture-arm,.compass-gesture-ring');if(!compassTouchSession&&!target)return;if(!compassTouchSession)compassTouchSession={pan:map.dragging.enabled(),zoom:map.touchZoom.enabled(),pinched:false};compassTouches.set(e.pointerId,{x:e.clientX,y:e.clientY});map.dragging.disable();map.touchZoom.disable();if(compassTouches.size>=2){const p=[...compassTouches.values()],c=compassState();snapshot();compassPinch={radiusM:c.radiusM,angle:c.angle,center:[...c.center],distance:Math.max(1,Math.hypot(p[1].x-p[0].x,p[1].y-p[0].y)),heading:gestureAngle(p[0],p[1]),mid:{x:(p[0].x+p[1].x)/2,y:(p[0].y+p[1].y)/2}};compassTouchSession.pinched=true;compassPointer=null;compassArc=[];compassFollowing=false;compassCenterMarker?.dragging?.disable();e.preventDefault();e.stopImmediatePropagation()}else if(target?.matches('.compass-gesture-arm,.compass-gesture-ring')){e.preventDefault();e.stopImmediatePropagation()}},true);
compassTouchHost.addEventListener('pointermove',e=>{if(!compassTouches.has(e.pointerId))return;compassTouches.set(e.pointerId,{x:e.clientX,y:e.clientY});if(!compassTouchSession?.pinched)return;e.preventDefault();e.stopImmediatePropagation();if(!compassPinch||compassTouches.size<2)return;const p=[...compassTouches.values()],c=compassState(),b=compassPinch;c.radiusM=Math.max(.1,Math.min(20000000,b.radiusM*Math.hypot(p[1].x-p[0].x,p[1].y-p[0].y)/b.distance));c.angle=b.angle+angleDelta(gestureAngle(p[0],p[1]),b.heading);if(!c.locked){const a=map.latLngToContainerPoint(b.center),mid={x:(p[0].x+p[1].x)/2,y:(p[0].y+p[1].y)/2},at=map.containerPointToLatLng([a.x+mid.x-b.mid.x,a.y+mid.y-b.mid.y]);c.center=[at.lat,at.lng,b.center[2]||0]}updateCompassGeometry();compassCenterMarker?.setLatLng(c.center)},true);
function endCompassTouch(e){if(!compassTouches.has(e.pointerId))return;const pinched=compassTouchSession?.pinched;compassTouches.delete(e.pointerId);if(pinched){e.preventDefault();e.stopImmediatePropagation();compassIgnoreClickUntil=performance.now()+600}if(compassTouches.size<2)compassPinch=null;if(!compassTouches.size){const session=compassTouchSession;compassTouchSession=null;if(session?.pan)map.dragging.enable();if(session?.zoom)map.touchZoom.enable();if(pinched){renderCompass();save()}}}
for(const type of ['pointerup','pointercancel'])compassTouchHost.addEventListener(type,endCompassTouch,true);
compassTouchHost.addEventListener('click',e=>{if(performance.now()<compassIgnoreClickUntil){e.preventDefault();e.stopImmediatePropagation()}},true);
