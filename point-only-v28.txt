function objectVertexName(o,index){return o.boxInputMode==='ordered'&&o.boxVertices?.[index]?.name||o.name+' · 頂點 '+(index+1)}
function inspectVertex(id,index){const found=getObject(id);if(!found)return;const key=id+':'+index;if(inspected===key){map.closePopup();selectObject(id);return}inspected=key;const o=found.object,content=showPointPopup(objectVertexName(o,index),o.points[index]),desc=document.createElement('p');desc.textContent=o.name+' · 高度 '+altitudeText(o.points[index][2]||0);const edit=document.createElement('button');edit.textContent='編輯此圖形';edit.onclick=()=>{map.closePopup();selectObject(id)};content.append(desc,edit);L.popup().setLatLng(o.points[index]).setContent(content).openOn(map)}
function bindUserPoint(marker,o,index){const el=marker.getElement();if(el){el.setAttribute('role','button');el.setAttribute('aria-label',o.type==='point'?o.name:objectVertexName(o,index));el.setAttribute('data-object-id',o.id);el.setAttribute('data-vertex-index',String(index))}marker.on('click',e=>{L.DomEvent.stopPropagation(e);if(tool==='erase')eraseObject(o.id);else if(tool==='pan'){if(o.type==='point')inspectObject(o.id);else inspectVertex(o.id,index)}});marker.on('dblclick',e=>{L.DomEvent.stopPropagation(e);clearTimeout(clickTimer);if(tool==='pan')selectObject(o.id)})}
function renderObjects(){draw.clearLayers();handles.clearLayers();objectPaths.clear();for(const layer of state.layers){if(!layer.visible)continue;for(const o of layer.objects){const alpha=layer.opacity??1,style={pane:o.type==='point'?'userPointPane':'userPane',interactive:o.type==='point',color:layer.color||o.color,opacity:alpha,weight:o.id===selected?4:2,fillColor:layer.color||o.color,fillOpacity:(o.type==='point'?.9:.1)*alpha,bubblingMouseEvents:false};const geometry=o.type==='point'?L.circleMarker(o.points[0],{...style,radius:7,className:'user-point'}):o.type==='polygon'?L.polygon(o.points,{...style,className:'noninteractive-region'}):L.polyline(o.points,style);geometry.addTo(draw).bindTooltip(o.name+(o.measureDistance?' · '+formatDistance(o.points.reduce((n,p,i)=>i?n+distance(o.points[i-1],p):n,0)):''),{permanent:o.measureDistance===true||layer.names===true,opacity:alpha});if(o.type==='point')bindUserPoint(geometry,o,0);else{const element=geometry.getElement();if(element){element.setAttribute('data-geometry-id',o.id);element.setAttribute('aria-label',o.name);element.setAttribute('aria-hidden','true')}const indices=o.type==='polygon'?o.points.map((_,i)=>i):[0,o.points.length-1];for(const index of indices){const point=L.circleMarker(o.points[index],{pane:'userPointPane',radius:5,weight:1.5,color:layer.color||o.color,fillColor:layer.color||o.color,fillOpacity:alpha,opacity:alpha,bubblingMouseEvents:false,className:'user-vertex-point'}).bindTooltip(objectVertexName(o,index),{permanent:false,opacity:alpha}).addTo(draw);bindUserPoint(point,o,index)}}objectPaths.set(o.id,geometry)}}renderInstruments()}
