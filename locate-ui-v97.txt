
function makeLocateButton(item){const button=document.createElement('button');button.type='button';button.textContent='定位';button.className='locate-toggle';button.dataset.locateKey=item.key;button.setAttribute('aria-pressed',String(locatedTarget?.key===item.key));button.onclick=e=>{e.stopPropagation();toggleLocate(item)};return button}
function syncLocationButtons(){for(const button of document.querySelectorAll('[data-locate-key]')){const active=locatedTarget?.key===button.dataset.locateKey;button.setAttribute('aria-pressed',String(active));button.title=active?'取消定位':'定位並標示';button.classList.toggle('active',active)}}
function revealWorkspaceItem(key){const row=[...document.querySelectorAll('[data-workspace-key]')].find(n=>n.dataset.workspaceKey===key);if(!row)return;document.body.classList.remove('collapsed');$('workspace-toggle').setAttribute('aria-expanded','true');for(let n=row;n&&n!==$('workspace');n=n.parentElement){if(n.tagName==='DETAILS')n.open=true;n.hidden=false}for(const old of document.querySelectorAll('.workspace-location'))old.classList.remove('workspace-location');row.classList.add('workspace-location');requestAnimationFrame(()=>{loadVisibleTables(row);row.scrollIntoView({block:'nearest',behavior:'smooth'});map.invalidateSize()})}
function toggleLocate(item){searchHighlights.clearLayers();if(locatedTarget?.key===item.key){locatedTarget=null;syncLocationButtons();for(const row of document.querySelectorAll('.workspace-location'))row.classList.remove('workspace-location');return}if(!item.points?.length)return toast('此物件沒有可定位座標');locatedTarget=item;syncLocationButtons();const color=state.settings.theme==='bright'?'#b65208':'#ffd27c';if(item.points.length===1){map.setView(item.points[0],Math.max(9,map.getZoom()));L.circleMarker(item.points[0],{radius:14,weight:3,color,fillOpacity:0,bubblingMouseEvents:false}).addTo(searchHighlights).on('click',()=>revealWorkspaceItem(item.key))}else{map.fitBounds(item.points,{padding:[45,45],maxZoom:11});L.polyline(item.points,{weight:8,color,opacity:.65,dashArray:'7 5',bubblingMouseEvents:false}).addTo(searchHighlights).on('click',()=>revealWorkspaceItem(item.key))}revealWorkspaceItem(item.key)}

function printAxisTicks(axes,pm){const b=pm.getBounds(),size=pm.getSize();for(const lat of [true,false]){const lo=lat?b.getSouth():b.getWest(),hi=lat?b.getNorth():b.getEast(),step=axisStep(hi-lo,Math.max(1,Math.floor((lat?size.y:size.x)/6)));let count=0;for(let v=Math.ceil(lo/step)*step;v<=hi&&count++<4000;v+=step){const p=pm.latLngToContainerPoint(lat?[v,pm.getCenter().lng]:[pm.getCenter().lat,v]),position=lat?p.y:p.x;if(position<2||position>(lat?size.y-24:size.x-2))continue;const tick=document.createElement('span'),seconds=Math.round(v*3600);tick.className='axis-tick '+(lat?'lat-tick':'lng-tick')+axisTickClass(seconds,step);tick.style[lat?'top':'left']=position+'px';if(!lat){tick.style.top='auto';tick.style.bottom='0'}axes.append(tick)}}}
